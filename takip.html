window.addEventListener("DOMContentLoaded", () => { map = L.map('map').setView([41.0082, 28.9784], 13); L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { maxZoom: 19, attribution: '© OpenStreetMap' }).addTo(map); // URL'den sipariş ID al (Örn: takip.html?id=123-abc) const urlParams = new URLSearchParams(window.location.search); const orderId = urlParams.get('id'); if (orderId) { loadAndTrackSpecificOrder(orderId); } else { loadAndTrackLatestOrder(); // Yedek olarak son sipariş } }); async function loadAndTrackSpecificOrder(orderId) { const { data: order, error } = await supabaseClient .from("orders") .select("*") .eq("id", orderId) .single(); if (error || !order) { alert("Sipariş bulunamadı!"); return; } updateUI(order); subscribeToOrder(order.id); } async function loadAndTrackLatestOrder() { const { data: orders, error } = await supabaseClient .from("orders") .select("*") .order("created_at", { ascending: false }) .limit(1); if (error || !orders.length) return; const order = orders[0]; updateUI(order); subscribeToOrder(order.id); } function subscribeToOrder(orderId) { supabaseClient .channel(`order-track-${orderId}`) .on('postgres_changes', { event: 'UPDATE', schema: 'public', table: 'orders', filter: `id=eq.${orderId}` }, payload => { updateUI(payload.new); }) .subscribe(); }
